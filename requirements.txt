# Pinned on purpose. A show machine gets set up once, weeks before the
# gig, and an unpinned file means the build that ships is whatever PyPI
# happened to be serving that morning — a detail nobody can reconstruct
# afterwards when a feed misbehaves on site. Every version below was
# checked to have a wheel for all three targets the project builds for:
# macOS arm64 on 3.12 (Syphon's ceiling), Windows x64 on 3.13 (Spout's),
# and Linux x86_64 on 3.12 for CI. None of them builds from source on a
# machine with no compiler, which is the other half of the point.
#
# To move them: bump one line, let CI and the macOS/Windows build jobs
# run, and keep the versions that come back green.

# Core (all platforms)
numpy==2.4.6
opencv-python==5.0.0.93
cyndilib==0.1.1          # NDI send/receive; bundles the NDI runtime, no SDK install needed

# Web control panel
fastapi==0.141.1
uvicorn==0.53.0
websockets==17.1

# CA certificates for activating shop licences over HTTPS: a packaged
# Python on macOS has no system bundle of its own.
#
# Pinned like the rest, but this one is a trust store rather than code:
# a frozen bundle goes stale as roots are added and withdrawn, and a
# stale one shows up as licence activation failing on a customer's
# machine months after the build. Refresh it deliberately at release
# time rather than leaving it to whatever the build host resolves.
certifi==2026.7.22

# Real camera names in the source picker (Windows/DirectShow)
pygrabber==0.2; sys_platform == "win32"

# Texture-share output (optional but recommended):
#   Syphon on macOS needs Python <= 3.12; Spout on Windows needs <= 3.13.
#   The Setup scripts pick a compatible Python automatically. Those
#   ceilings are real: neither project publishes a wheel above them, so
#   a newer Python silently loses texture-share output.
syphon-python==0.1.1; sys_platform == "darwin" and python_version < "3.13"
SpoutGL==0.1.1; sys_platform == "win32" and python_version < "3.14"

# ONNX Runtime: the Selfie and MODNet silhouettes, the CenterFace
# detector, and RVM in internal builds.
#
#   macOS: the plain wheel carries the CoreML provider, so the silhouette
#   models reach the Apple GPU / Neural Engine with nothing extra; the
#   default YuNet detector needs nothing at all.
#   Linux: not a target for the app, but CI runs there and without this
#   every ONNX test skips — which is how a provider bug would reach a tag
#   unnoticed.
#   Windows: the GPU runtime depends on the card and is not pinned here,
#   because onnxruntime-gpu (CUDA, NVIDIA) and onnxruntime-directml (any
#   DirectX 12 card: AMD, Intel, NVIDIA) are the same package underneath
#   and cannot both be installed. The launcher looks for an NVIDIA card
#   and installs requirements-nvidia.txt or requirements-directml.txt;
#   the release workflow installs the NVIDIA one for the installer.
#   Either runtime advertises providers it cannot necessarily load (a
#   CUDA build on a machine with no CUDA fails with "cublas64_12.dll is
#   missing"), and yewee treats that as a fallback to the CPU rather than
#   an error (see runtime.make_session). `--doctor` says which you have.
onnxruntime==1.30.0; sys_platform == "darwin" or sys_platform == "linux"
